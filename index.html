<script>

/* ── App state ── */

let A = { code: null, className: null, name: null, teacher: false, iv: null };


/* ── Navigation ── */

function go(id) {

  document.querySelectorAll('.section').forEach(s => s.classList.remove('active'));

  document.getElementById(id).classList.add('active');

}


/* ── Utilities ── */

function esc(s) {

  return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');

}

function timeAgo(ts) {

  const s = Math.floor((Date.now()-ts)/1000);

  if (s < 60) return s+'s ago';

  const m = Math.floor(s/60);

  if (m < 60) return m+'m ago';

  return Math.floor(m/60)+'h ago';

}


/* ── Storage ── */

function qKey(c)   { return 'chq_q_'+(c||'').toUpperCase(); }

function cfgKey(c) { return 'chq_c_'+(c||'').toUpperCase(); }

function getQ(c)   { try { return JSON.parse(localStorage.getItem(qKey(c))||'[]'); } catch { return []; } }

function setQ(c,q) { localStorage.setItem(qKey(c), JSON.stringify(q)); }

function getCfg(c) { try { return JSON.parse(localStorage.getItem(cfgKey(c))); } catch { return null; } }

function setCfg(c,v){ localStorage.setItem(cfgKey(c.toUpperCase()), JSON.stringify(v)); }


/* ── Student: check class code ── */

function sCheckCode() {

  const raw = document.getElementById('i-scode').value.trim().toUpperCase();

  const err = document.getElementById('e-scode');

  if (!raw) { err.textContent = 'Enter your class code.'; return; }

  const cfg = getCfg(raw);

  if (!cfg) { err.textContent = `No class found for "${raw}". Double-check with your teacher.`; return; }

  A.code = raw; A.className = cfg.className;

  err.textContent = '';

  document.getElementById('sname-class-label').textContent = 'Class: ' + cfg.className;

  go('s-sname');

  setTimeout(() => document.getElementById('i-sname').focus(), 80);

}


/* ── Student: submit name ── */

function sEnter() {

  const n = document.getElementById('i-sname').value.trim();

  const err = document.getElementById('e-sname');

  if (!n) { err.textContent = 'Enter your first name.'; return; }

  A.name = n;

  err.textContent = '';

  document.getElementById('smain-class').textContent = A.className;

  document.getElementById('smain-name').textContent = n;

  go('s-smain');

  poll(renderStudent);

  renderStudent();

}


function sLogout() {

  Object.assign(A, { code:null, className:null, name:null });

  stopPoll();

  document.getElementById('i-sname').value = '';

  document.getElementById('i-scode').value = '';

  go('s-land');

}


/* ── Student: join / leave ── */

function sJoin() {

  const q = getQ(A.code);

  if (!q.find(x => x.name.toLowerCase()===A.name.toLowerCase())) {

    q.push({ name: A.name, ts: Date.now() });

    setQ(A.code, q);

  }

  renderStudent();

}


function sLeave() {

  setQ(A.code, getQ(A.code).filter(x => x.name.toLowerCase()!==A.name.toLowerCase()));

  renderStudent();

}


/* ── Student: render ── */

function renderStudent() {

  const q = getQ(A.code);

  const idx = q.findIndex(x => x.name.toLowerCase()===A.name.toLowerCase());

  const inQ = idx !== -1;

  const sb = document.getElementById('s-status');

  const pi = document.getElementById('s-pos');

  const jb = document.getElementById('s-join');

  const lb = document.getElementById('s-leave');


  if (inQ) {

    sb.className = idx===0 ? 'status-bar s-next' : 'status-bar s-waiting';

    sb.textContent = idx===0 ? "🎉 You're next! Get ready." : "⏳ You're in the queue";

    pi.style.display = 'block';

    document.getElementById('s-pos-n').textContent = idx===0 ? '🎉' : '#'+(idx+1);

    document.getElementById('s-pos-t').textContent = idx===0 ? "You're next!" : ' of '+q.length+' in line';

    jb.style.display = 'none'; lb.style.display = 'block';

  } else {

    sb.className = 'status-bar s-idle';

    sb.textContent = 'Not in queue — click to join when you need help';

    pi.style.display = 'none';

    jb.style.display = 'block'; lb.style.display = 'none';

  }


  document.getElementById('s-count').textContent = q.length;

  const el = document.getElementById('s-list');


  if (!q.length) { el.innerHTML='<div class="empty">✅ Queue is empty — no wait!</div>'; return; }


  el.innerHTML = q.map((item,i) => {

    const isYou = item.name.toLowerCase()===A.name.toLowerCase();

    return `<div class="queue-item ${i===0?'first':''}">

      <span class="q-num">${i+1}</span>

      <span class="q-name">${esc(item.name)}</span>

      ${i===0 ? '<span class="badge badge-next">Next</span>' : ''}

      ${isYou&&i>0 ? '<span class="badge badge-you">You</span>' : ''}

      <span class="q-time">${timeAgo(item.ts)}</span>

    </div>`;

  }).join('');

}


/* ── Teacher: login ── */

function tLogin() {

  const code = document.getElementById('i-tcode').value.trim().toUpperCase();

  const pass = document.getElementById('i-tpass').value;

  const err  = document.getElementById('e-tlogin');

  if (!code) { err.textContent = 'Enter your class code.'; return; }

  if (!pass) { err.textContent = 'Enter your passcode.'; return; }

  const cfg = getCfg(code);

  if (!cfg)            { err.textContent = `No class found for "${code}". Set up your class first.`; return; }

  if (cfg.passcode !== pass) { err.textContent = 'Wrong passcode.'; return; }

  err.textContent = '';

  A.code = code; A.className = cfg.className; A.teacher = true;

  document.getElementById('t-classname').textContent = cfg.className;

  document.getElementById('t-codebox').textContent = code;

  go('s-tdash');

  poll(renderTeacher);

  renderTeacher();

}


/* ── Teacher: setup ── */

function tSetup() {

  const cn   = document.getElementById('i-tclassname').value.trim();

  const code = document.getElementById('i-tclasscode').value.trim().toUpperCase();

  const p1   = document.getElementById('i-tpass1').value;

  const p2   = document.getElementById('i-tpass2').value;

  const err  = document.getElementById('e-tsetup');

  if (!cn)   { err.textContent = 'Enter a class name.'; return; }

  if (!code) { err.textContent = 'Enter a class code.'; return; }

  if (!/^[A-Z0-9]{2,10}$/.test(code)) { err.textContent = 'Code must be 2–10 letters/numbers, no spaces.'; return; }

  if (!p1)   { err.textContent = 'Choose a passcode.'; return; }

  if (p1!==p2){ err.textContent = "Passcodes don't match."; return; }

  setCfg(code, { className: cn, passcode: p1 });

  err.textContent = '';

  document.getElementById('i-tcode').value = code;

  document.getElementById('i-tpass').value = p1;

  go('s-tlogin');

  setTimeout(() => alert(`✅ Class "${cn}" created!\n\nStudent class code: ${code}\n\nYou're ready to log in.`), 80);

}


/* ── Teacher: actions ── */

function tRemove(name) {

  setQ(A.code, getQ(A.code).filter(x => x.name !== name));

  renderTeacher();

}

function tClearAll() {

  if (!confirm('Clear the entire queue?')) return;

  setQ(A.code, []); renderTeacher();

}

function tLogout() {

  A.teacher = false; A.code = null; stopPoll();

  document.getElementById('i-tpass').value = '';

  document.getElementById('i-tcode').value = '';

  go('s-land');

}


/* ── Teacher: render ── */

function renderTeacher() {

  const q = getQ(A.code);

  document.getElementById('t-count').textContent = q.length;

  const banner = document.getElementById('t-banner');

  if (q.length) {

    banner.style.display = 'flex';

    document.getElementById('t-next').textContent = q[0].name;

  } else {

    banner.style.display = 'none';

  }

  const el = document.getElementById('t-list');

  if (!q.length) { el.innerHTML='<div class="empty">✅ Queue is empty. All students are set!</div>'; return; }

  el.innerHTML = q.map((item,i) => `

    <div class="teacher-item ${i===0?'first':''}">

      <span style="font-family:'DM Mono',monospace;font-size:11px;color:var(--muted);min-width:20px">${i+1}</span>

      <span class="tname">${esc(item.name)}</span>

      <span style="font-size:11px;color:var(--muted);font-family:'DM Mono',monospace">${timeAgo(item.ts)}</span>

      <button class="done-btn" onclick="tRemove(${JSON.stringify(item.name)})">Done ✓</button>

    </div>`).join('');

}


/* ── Polling ── */

function poll(fn) { stopPoll(); A.iv = setInterval(fn, 2500); }

function stopPoll() { if (A.iv) { clearInterval(A.iv); A.iv = null; } }


/* Cross-tab sync via storage event */

window.addEventListener('storage', () => {

  if (A.teacher) renderTeacher();

  else if (A.name) renderStudent();

});
